######### DOM :Document object model : is a way js represent html webpage so that js can access and change its elements
->Four pillar of dom 
  ->1:Selection of an element 
  ->2:Changing HTML 
  ->3:Changing css
  ->4:Event Listner 

### Selection of an element
var h1 =documnet.querySelector('hi') # This selects the first <h1> element.
var idd = documnet.querySelector(#id_name) # for id
var clsss=documnet.querySelector(.class_name) # for  class 

### Changing of html

var a = documnet.querySelector('h1')
a.innerHTML="change"    # to change innerhtml 

### Changing css 
a.style.color='black' # to change css a.style.property
